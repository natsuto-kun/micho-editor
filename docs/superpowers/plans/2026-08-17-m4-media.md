> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** JPEG/PNG/WebP 画像をエディタにドラッグ&ドロップすると `![[media:id]]` 構文が挿入され、ライブプレビューで `<img>` に変換される。

**Tech Stack:** Go `golang.org/x/image`（リサイズ・WebP デコード）、Wails v2 `OnFileDrop` / `AssetServer`、CodeMirror 6 `WidgetType` / `RangeSetBuilder`

```go
package media

import (
	"bytes"
	"database/sql"
	"fmt"
	"image"
	"image/color"
	stdraw "image/draw"
	"image/jpeg"
	_ "image/png"
	"net/http"
	"os"
	"path/filepath"
	"time"

	"github.com/google/uuid"
	xdraw "golang.org/x/image/draw"
	_ "golang.org/x/image/webp"
)

const (
	maxLong = 1200
	quality = 85
)

// Record represents a stored media file.
type Record struct {
	ID         string `json:"id"`
	ScenarioID string `json:"scenarioId"`
	Filename   string `json:"filename"`
	Mime       string `json:"mime"`
	Width      int    `json:"width"`
	Height     int    `json:"height"`
	Bytes      int    `json:"bytes"`
	RelPath    string `json:"relPath"`
}

// ImportFromPath imports an image file from a local file path.
func ImportFromPath(db *sql.DB, mediaDir, scenarioID, srcPath string) (Record, error) {
	data, err := os.ReadFile(srcPath)
	if err != nil {
		return Record{}, fmt.Errorf("media: read %q: %w", srcPath, err)
	}
	return ImportFromBytes(db, mediaDir, scenarioID, filepath.Base(srcPath), data)
}

// ImportFromBytes imports image bytes into media storage.
// Supported MIME types: image/jpeg, image/png, image/webp.
// All images are resized to a max long edge of 1200px and saved as JPEG 85%.
func ImportFromBytes(db *sql.DB, mediaDir, scenarioID, filename string, data []byte) (Record, error) {
	sniff := data
	if len(sniff) > 512 {
		sniff = sniff[:512]
	}
	mime := http.DetectContentType(sniff)
	switch mime {
	case "image/jpeg", "image/png", "image/webp":
	default:
		return Record{}, fmt.Errorf("media: unsupported type %q", mime)
	}

	src, _, err := image.Decode(bytes.NewReader(data))
	if err != nil {
		return Record{}, fmt.Errorf("media: decode: %w", err)
	}

	img := flattenAndResize(src)
	b := img.Bounds()
	w, h := b.Dx(), b.Dy()

	id := uuid.New().String()

	if err := os.MkdirAll(mediaDir, 0o755); err != nil {
		return Record{}, fmt.Errorf("media: mkdir: %w", err)
	}
	fullPath := filepath.Join(mediaDir, id+".jpg")

	var buf bytes.Buffer
	if err := jpeg.Encode(&buf, img, &jpeg.Options{Quality: quality}); err != nil {
		return Record{}, fmt.Errorf("media: jpeg encode: %w", err)
	}
	if err := os.WriteFile(fullPath, buf.Bytes(), 0o644); err != nil {
		return Record{}, fmt.Errorf("media: write file: %w", err)
	}
	sz := buf.Len()

	relPath := filepath.Join("media", id+".jpg")
	_, err = db.Exec(
		`INSERT INTO media (id, scenario_id, filename, mime, width, height, bytes, rel_path, created_at)
		 VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`,
		id, scenarioID, filename, "image/jpeg", w, h, sz, relPath, time.Now().Unix(),
	)
	if err != nil {
		return Record{}, fmt.Errorf("media: db insert: %w", err)
	}

	return Record{
		ID:         id,
		ScenarioID: scenarioID,
		Filename:   filename,
		Mime:       "image/jpeg",
		Width:      w,
		Height:     h,
		Bytes:      sz,
		RelPath:    relPath,
	}, nil
}

// List returns all media records for a scenario, ordered by created_at.
func List(db *sql.DB, scenarioID string) ([]Record, error) {
	rows, err := db.Query(
		`SELECT id, scenario_id, filename, mime, width, height, bytes, rel_path
		 FROM media WHERE scenario_id = ? ORDER BY created_at`,
		scenarioID,
	)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var out []Record
	for rows.Next() {
		var r Record
		if err := rows.Scan(&r.ID, &r.ScenarioID, &r.Filename, &r.Mime, &r.Width, &r.Height, &r.Bytes, &r.RelPath); err != nil {
			return nil, err
		}
		out = append(out, r)
	}
	return out, rows.Err()
}

// flattenAndResize composites the image on a white background (to handle alpha)
// and resizes it so the long edge is at most maxLong pixels.
func flattenAndResize(src image.Image) image.Image {
	b := src.Bounds()
	flat := image.NewRGBA(image.Rect(0, 0, b.Dx(), b.Dy()))
	// White background
	stdraw.Draw(flat, flat.Bounds(), image.NewUniform(color.White), image.Point{}, stdraw.Src)
	// Source on top
	stdraw.Draw(flat, flat.Bounds(), src, b.Min, stdraw.Over)
	return scaleDown(flat)
}

// scaleDown reduces the image so its long edge is at most maxLong pixels.
// Returns the original if it already fits.
func scaleDown(src *image.RGBA) image.Image {
	b := src.Bounds()
	w, h := b.Dx(), b.Dy()
	if w <= maxLong && h <= maxLong {
		return src
	}
	var nw, nh int
	if w >= h {
		nw = maxLong
		nh = max(1, h*maxLong/w)
	} else {
		nh = maxLong
		nw = max(1, w*maxLong/h)
	}
	dst := image.NewRGBA(image.Rect(0, 0, nw, nh))
	xdraw.CatmullRom.Scale(dst, dst.Bounds(), src, b, xdraw.Over, nil)
	return dst
}
```

```go
package media

import (
	"database/sql"
	"net/http"
	"os"
	"path/filepath"
	"strings"
)

// Handler serves media files over HTTP for the Wails AssetServer.
// Mount at any path; it responds to /media/{id} and returns 404 otherwise.
type Handler struct {
	db     *sql.DB
	appDir string // root directory (rel_path values are relative to this)
}

// New creates a Handler that serves media from appDir using records in db.
func New(db *sql.DB, appDir string) *Handler {
	return &Handler{db: db, appDir: appDir}
}

func (h *Handler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
	if !strings.HasPrefix(r.URL.Path, "/media/") {
		http.NotFound(w, r)
		return
	}
	id := strings.TrimPrefix(r.URL.Path, "/media/")
	// Reject empty IDs or path traversal attempts
	if id == "" || strings.ContainsAny(id, "/\\") {
		http.NotFound(w, r)
		return
	}

	var relPath string
	err := h.db.QueryRow(`SELECT rel_path FROM media WHERE id = ?`, id).Scan(&relPath)
	if err == sql.ErrNoRows {
		http.NotFound(w, r)
		return
	}
	if err != nil {
		http.Error(w, "internal server error", http.StatusInternalServerError)
		return
	}

	fullPath := filepath.Join(h.appDir, relPath)
	f, err := os.Open(fullPath)
	if err != nil {
		http.NotFound(w, r)
		return
	}
	defer f.Close()

	info, err := f.Stat()
	if err != nil {
		http.Error(w, "internal server error", http.StatusInternalServerError)
		return
	}
	// http.ServeContent sets Content-Type from the filename extension (.jpg → image/jpeg)
	http.ServeContent(w, r, id+".jpg", info.ModTime(), f)
}
```

```go
func (a *App) serveMedia(w http.ResponseWriter, r *http.Request) {
	if a.mediaHandler == nil {

}
```

```typescript
export interface MediaImageMatch {
  from: number;
  to: number;
  mediaId: string;
}

// Matches ![[media:{uuid}]] inline syntax.
// UUID characters: alphanumeric and hyphens.
const MEDIA_RE = /!\[\[media:([a-zA-Z0-9-]+)\]\]/g;

/**
 * Scans `text` (which starts at document position `offset`) for all
 * ![[media:id]] occurrences and returns their document positions and IDs.
 */
export function findMediaImages(text: string, offset: number): MediaImageMatch[] {
  const results: MediaImageMatch[] = [];
  const re = new RegExp(MEDIA_RE.source, "g");
  let m: RegExpExecArray | null;
  while ((m = re.exec(text)) !== null) {
    results.push({
      from: offset + m.index,
      to: offset + m.index + m[0].length,
      mediaId: m[1],
    });
  }
  return results;
}
```

```typescript
import { WidgetType, EditorView } from "@codemirror/view";

export class ImageWidget extends WidgetType {
  constructor(readonly mediaId: string) {
    super();
  }

  eq(other: WidgetType): boolean {
    return other instanceof ImageWidget && other.mediaId === this.mediaId;
  }

  toDOM(_view: EditorView): HTMLElement {
    const img = document.createElement("img");
    img.src = `/media/${this.mediaId}`;
    img.loading = "lazy";
    img.className = "media-image";
    img.alt = `media:${this.mediaId}`;
    img.style.maxWidth = "100%";
    img.style.display = "block";
    img.style.margin = "4px 0";
    return img;
  }

  ignoreEvent(_event: Event): boolean {
    return false;
  }
}
```

- [ ] **Step 2: `buildDecorations` 関数を修正**

現在の `buildDecorations` は `directive` ブロックのみを扱っている。画像デコレーションを追加するため、`RangeSetBuilder` への追加方式を「事前にすべて収集 → ソート → 追加」に変更する。

`buildDecorations` 関数全体を以下で置き換える:

```typescript
type DecoEntry = { from: number; to: number; deco: Decoration };

function buildDecorations(view: EditorView): DecorationSet {
  const cursor = view.state.selection.main.head;
  const doc = view.state.doc;
  const entries: DecoEntry[] = [];

  // --- ディレクティブブロック ---
  const blocks = parseDirectiveBlocks(doc, view.visibleRanges);

  for (const block of blocks) {
    const openLine = doc.lineAt(block.from);
    const closeLine = doc.lineAt(block.to);

    const cursorOnOpen = cursor >= openLine.from && cursor <= openLine.to;
    const cursorOnClose = cursor >= closeLine.from && cursor <= closeLine.to;

    entries.push({
      from: openLine.from,
      to: openLine.from,
      deco: Decoration.line({ class: `directive-open directive-${block.type}-open` }),
    });
    if (!cursorOnOpen) {
      const openEnd = Math.min(openLine.to + 1, doc.length);
      entries.push({
        from: openLine.from,
        to: openEnd,
        deco: Decoration.replace({
          widget: new DirectiveHeaderWidget(block.type, block.params),
          block: true,
        }),
      });
    }

    for (let n = openLine.number + 1; n <= closeLine.number - 1; n++) {
      const line = doc.line(n);
      entries.push({
        from: line.from,
        to: line.from,
        deco: Decoration.line({ class: `directive-body directive-${block.type}-body` }),
      });
    }

    entries.push({
      from: closeLine.from,
      to: closeLine.from,
      deco: Decoration.line({ class: `directive-close directive-${block.type}-close` }),
    });
    if (!cursorOnClose) {
      const closeEnd = Math.min(closeLine.to + 1, doc.length);
      entries.push({
        from: closeLine.from,
        to: closeEnd,
        deco: Decoration.replace({
          widget: new DirectiveFooterWidget(block.type),
          block: true,
        }),
      });
    }
  }

  // --- メディア画像インライン ---
  for (const range of view.visibleRanges) {
    const text = doc.sliceString(range.from, range.to);
    const images = findMediaImages(text, range.from);
    for (const img of images) {
      const cursorIn = cursor >= img.from && cursor <= img.to;
      if (!cursorIn) {
        entries.push({
          from: img.from,
          to: img.to,
          deco: Decoration.replace({ widget: new ImageWidget(img.mediaId) }),
        });
      }
    }
  }

  // RangeSetBuilder は from の昇順で追加する必要がある
  entries.sort((a, b) => a.from - b.from || a.to - b.to);

  const builder = new RangeSetBuilder<Decoration>();
  for (const { from, to, deco } of entries) {
    builder.add(from, to, deco);
  }
  return builder.finish();
}
```

```typescript
// Handle drag-and-drop of image files onto the editor area.
useEffect(() => {
  OnFileDrop(async (x, y, paths) => {
    const view = viewRef.current;
    const sid = useScenarioStore.getState().scenarioId;
    if (!view || !sid) return;

    // Only process drops within the editor's DOM bounds.
    const editorRect = view.dom.getBoundingClientRect();
    if (
      x < editorRect.left ||
      x > editorRect.right ||
      y < editorRect.top ||
      y > editorRect.bottom
    ) return;

    const path = paths[0];
    if (!path) return;
    const ext = path.slice(path.lastIndexOf(".")).toLowerCase();
    if (![".jpg", ".jpeg", ".png", ".webp"].includes(ext)) return;

    try {
      const record = await API.importMediaFromPath(sid, path);
      const pos = view.posAtCoords({ x, y }) ?? view.state.doc.length;
      view.dispatch({
        changes: { from: pos, insert: `![[media:${record.id}]]` },
      });
    } catch (err) {
      console.error("Media import failed:", err);
    }
  }, false);

  return () => OnFileDropOff();
}, []);
```
